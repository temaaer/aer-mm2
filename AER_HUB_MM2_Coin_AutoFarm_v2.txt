--[[ AER HUB MM2 COIN AUTO FARM ]]

--[[
    AER HUB MM2 Coin Auto Farm - standalone script
    Core coin-farming algorithms are extracted from the user-provided pulsehub.txt.
    Standalone runtime + lightweight custom UI. FarmState/FarmMotion are from the supplied Pulse Hub source.
    Adds full-bag respawn, murderer-aware coin selection, and collision restoration.

    Methods: Under Map / Fast / Teleport
    Requires a Roblox executor that supports firetouchinterest and getgenv.
]]

local CollectionService = game:GetService("CollectionService")
local HttpService = game:GetService("HttpService")
local LocalPlayer = Players.LocalPlayer
local Workspace = game:GetService("Workspace")
local GlobalEnv = (type(getgenv) == "function" and getgenv()) or _G

    game = game,
    workspace = Workspace,
    Instance = Instance,
    Vector3 = Vector3,
    CFrame = CFrame,
    Enum = Enum,

    firetouchinterest = firetouchinterest,

local context = {
    environment = environment,
    workspace = Workspace,
    collection = CollectionService,
    players = Players,
    player = LocalPlayer,
    runService = RunService,
    storage = game:GetService("ReplicatedStorage"),
    http = HttpService,
    configuration = {farm = {speed = 21, fastSpeed = 24, tpSpeed = 25, method = "Under Map"}},
    isAlive = function() return true end,
}

local FarmState = {}
local FARM_ROLES = {Zombie = true, Survivor = true, Vampire = true, Hunter = true, Villager = true, Freezer = true, Runner = true}
local FARM_OPTIONS = {"respawnOnFull", "antiFling", "avoidMurder", "autoFling", "killAura", "pickGun", "shootOnFull"}

function FarmState.create(context)
	local state = {enabled = false, round = false, active = false, session = 0, total = 0, rate = 0,
		respawnOnFull = false, antiFling = false, avoidMurder = false, autoFling = false, killAura = true,
		pickGun = true, shootOnFull = true, collected = {}, full = {}, skipped = {}, containers = {}, containersAt = 0,
		bagItems = {}, bagAt = 0, coinsStarted = 0, knownRound = false, forcedFull = false}
	local environment = context.environment
	pcall(function()
		if not environment.isfile or not environment.readfile or not environment.isfile("PulseFarmMM2_config.json") then return end
		local saved = context.http:JSONDecode(environment.readfile("PulseFarmMM2_config.json"))
		if type(saved) ~= "table" then return end
		for _, key in FARM_OPTIONS do if type(saved[key]) == "boolean" then state[key] = saved[key] end end
		if type(saved.totalCoins) == "number" then state.total = saved.totalCoins end
	end)
	return state
end

function FarmState.save(context, state)
	if type(context.environment.writefile) ~= "function" then return end
	pcall(function()
		local saved = {totalCoins = state.total}
		for _, key in FARM_OPTIONS do saved[key] = state[key] end
		context.environment.writefile("PulseFarmMM2_config.json", context.http:JSONEncode(saved))
	end)
end

function FarmState.roundStart(state, now, coinsStarted)
	state.round, state.skipped = true, {}
	if not coinsStarted then state.collected = {} end
	table.clear(state.full)
	state.forcedFull, state.coinsStarted, state.knownRound = false, coinsStarted and now or 0, true
	if not coinsStarted then state.active = false end
end

function FarmState.roundEnd(state)
	state.round, state.active, state.collected, state.forcedFull, state.coinsStarted = false, false, {}, false, 0
end

function FarmState.coinCollected(state, now, kind, count, maximum)
	if state.coinsStarted == 0 then state.coinsStarted = now end
	if type(kind) == "string" and type(count) == "number" and type(maximum) == "number" then state.full[kind] = count >= maximum end
	local name = type(kind) == "string" and kind or "Coin"
	local previous = state.collected[name] or 0
	if not state.knownRound and state.collected[name] == nil and type(count) == "number" and count > 1 then previous = count - 1 end
	local added = 1
	if type(count) == "number" then
		added = count >= previous and count - previous or count
		state.collected[name] = count
	end
	if added > 0 then state.session, state.total, state.dirty = state.session + added, state.total + added, true end
	return added
end

function FarmState.refreshBag(context, state)
	local gui = context.player:FindFirstChild("PlayerGui")
	local main = gui and gui:FindFirstChild("MainGUI")
	local game = main and main:FindFirstChild("Game")
	state.bag, state.bagItems = game and game:FindFirstChild("CoinBags") or nil, {}
	if not state.bag then return end
	for _, item in state.bag:GetDescendants() do
		if item.Name == "Full" and item:IsA("GuiObject") and not item:FindFirstAncestorWhichIsA("LuaSourceContainer") then
			table.insert(state.bagItems, item)
		end
	end
end

function FarmState.isFull(context, state, now)
	if state.forcedFull then return true end
	if state.coinsStarted == 0 then return false end
	local any, all = false, true
	for _, full in state.full do any = true; if not full then all = false end end
	if any and all then return true end
	if now - state.coinsStarted < 2 then return false end
	if not state.bag or not state.bag.Parent or now - state.bagAt > 2 then
		state.bagAt = now
		FarmState.refreshBag(context, state)
	end
	if not state.bag or not state.bag.Visible then return false end
	for _, item in state.bagItems do if item.Parent and item.Visible then return true end end
	return false
end

function FarmState.validCoin(context, state, coin)
	if context.environment.getgenv().__pulseAntiCoin or not coin or not coin.Parent or not coin:IsA("BasePart") then return false end
	if not coin:FindFirstChild("TouchInterest") or coin:GetAttribute("Collected") or state.skipped[coin] then return false end
	local id = coin:GetAttribute("CoinID")
	if id ~= nil and state.full[id] then return false end
	return true
end

function FarmState.containers(context, state, now)
	if now - state.containersAt < 0.5 then return state.containers end
	state.containersAt = now
	local containers, seen = {}, {}
	for _, object in context.workspace:GetChildren() do
		local container = object:FindFirstChild("CoinContainer")
		if container and not seen[container] then seen[container] = true; table.insert(containers, container) end
	end
	for _, coin in context.collection:GetTagged("ServerCoinPart") do
		local container = coin.Parent
		if container and not seen[container] and container ~= context.workspace and container:IsDescendantOf(context.workspace) then
			seen[container] = true
			table.insert(containers, container)
		end
	end
	state.containers = containers
	return containers
end

function FarmState.canFarm(context, state, roles)
	if context.environment.getgenv().__pulseRoleFarmHold or type(roles) ~= "table" then return false end
	local role = roles[context.player.Name]
	if not role or not role.Role then return false end
	if role.Dead ~= true and role.Killed ~= true then return true end
	local character = context.player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	return FARM_ROLES[tostring(role.Role)] == true and humanoid ~= nil and humanoid.Health > 0
end


local FarmMotion = {}

function FarmMotion.create()
	return {phase = "idle", gotT = 0, learn = {}, tries = 0, got = 0, fails = 0, info = {}, skip = {}, pending = {},
		att = {}, gapT = 4.5, tp = {}, containerAt = 0, containers = {}, filterAt = 0}
end

function FarmMotion.parts(character)
	local parts = {}
	if not character then return parts end
	local head = character:FindFirstChild("Head")
	if head then table.insert(parts, head) end
	for _, child in character:GetChildren() do
		if not child:IsA("Accessory") then continue end
		local handle = child:FindFirstChild("Handle")
		if handle then table.insert(parts, handle) end
	end
	for _, name in {"UpperTorso", "Torso", "HumanoidRootPart", "LowerTorso"} do
		local part = character:FindFirstChild(name)
		if part then table.insert(parts, part) end
	end
	return parts
end

function FarmMotion.touch(context, character, coin)
	local environment = context.environment
	if type(environment.firetouchinterest) ~= "function" or not character or not coin or not coin.Parent then return end
	local globals = environment.getgenv()
	if globals.__pulseInvis and globals.__pulseInvis.active then globals.__pulseInvisPause = environment.os.clock() + 0.35 end
	for _, part in FarmMotion.parts(character) do
		pcall(function() environment.firetouchinterest(part, coin, 0); environment.firetouchinterest(part, coin, 1) end)
	end
end

function FarmMotion.containers(context, state, motion, now)
	if now - motion.containerAt > 1 then
		motion.containerAt = now
		motion.containers = FarmState.containers(context, state, now)
	end
	return motion.containers
end

function FarmMotion.rayParams(context, state, motion, now)
	if not motion.rayParams then
		motion.rayParams = context.environment.RaycastParams.new()
		motion.rayParams.FilterType = context.environment.Enum.RaycastFilterType.Exclude
		pcall(function() motion.rayParams.RespectCanCollide = true end)
	end
	if now - motion.filterAt > 1 then
		motion.filterAt = now
		local ignored = {}
		for _, player in context.players:GetPlayers() do if player.Character then table.insert(ignored, player.Character) end end
		for _, container in FarmMotion.containers(context, state, motion, now) do table.insert(ignored, container) end
		motion.rayParams.FilterDescendantsInstances = ignored
	end
	return motion.rayParams
end

function FarmMotion.floor(context, state, motion, position, offset, depth, now)
	local vector = context.environment.Vector3
	local hit = context.workspace:Raycast(position + vector.new(0, offset or 0.5, 0), vector.new(0, -(depth or 50), 0),
		FarmMotion.rayParams(context, state, motion, now))
	return hit and hit.Position.Y or nil
end

function FarmMotion.coinInfo(context, state, motion, coin, now)
	if motion.info[coin] then return motion.info[coin] end
	local position = coin.Position
	local surface = FarmMotion.floor(context, state, motion, position, 0.5, 40, now) or position.Y - 3
	local hide = context.environment.Vector3.new(position.X, surface - 3.8, position.Z)
	local info = {surf = surface, hide = hide, exposed = FarmMotion.floor(context, state, motion, hide, 0, 7, now) ~= nil}
	motion.info[coin] = info
	return info
end

function FarmMotion.murderPenalty(state, position, murderer)
	if not murderer then return 0 end
	local distance = (position - murderer).Magnitude
	if distance < 24 then return state.avoidMurder and 1000000 or 300 end
	if distance < 48 then return (48 - distance) * 2 end
	return 0
end

function FarmMotion.choose(context, state, motion, position, murderer, now)
	local chosen, lowest
	for _, container in FarmMotion.containers(context, state, motion, now) do
		if not container.Parent then continue end
		for _, coin in container:GetChildren() do
			if not FarmState.validCoin(context, state, coin) or (motion.skip[coin] or 0) >= now or motion.pending[coin] then continue end
			local info = FarmMotion.coinInfo(context, state, motion, coin, now)
			local score = (info.hide - position).Magnitude + math.max(0, coin.Position.Y - info.hide.Y) * 1.2
			if info.exposed then score += 25 end
			score += FarmMotion.murderPenalty(state, coin.Position, murderer)
			if not lowest or score < lowest then chosen, lowest = coin, score end
		end
	end
	if chosen and state.avoidMurder and lowest >= 1000000 then return nil end
	return chosen, lowest
end

function FarmMotion.jump(context, motion, root, destination, forced, now)
	if not forced and (destination - root.Position).Magnitude <= 1000 then return false end
	root.CFrame = context.environment.CFrame.new(destination)
	root.AssemblyLinearVelocity, root.AssemblyAngularVelocity = context.environment.Vector3.zero, context.environment.Vector3.zero
	motion.goT, motion.jumps = now, (motion.jumps or 0) + 1
	return true
end

function FarmMotion.hideAt(context, state, motion, root, now)
	local floor = FarmMotion.floor(context, state, motion, root.Position, 9, 30, now)
	if not floor then return root.Position end
	local position = root.Position
	return context.environment.Vector3.new(position.X, math.min(position.Y, floor - 3.8), position.Z)
end

function FarmMotion.canShootAt(context, position, root)
	local environment, workspace = context.environment, context.workspace
	local character = root.Parent
	local head = character and character:FindFirstChild("Head")
	if not head then return true end
	local attachment = root:FindFirstChild("GunRaycastAttachment")
	local ray = environment.RaycastParams.new()
	ray.FilterType, ray.FilterDescendantsInstances = environment.Enum.RaycastFilterType.Exclude, {character}
	if not attachment then
		local overlap = environment.OverlapParams.new()
		overlap.FilterType, overlap.FilterDescendantsInstances = environment.Enum.RaycastFilterType.Exclude, {character}
		return #workspace:GetPartBoundsInBox(environment.CFrame.new(position + environment.Vector3.new(0, 1, 0)),
			environment.Vector3.new(5, 4.5, 5), overlap) == 0
	end
	local vector, frame = environment.Vector3, environment.CFrame
	local headLocal, muzzle = root.CFrame:PointToObjectSpace(head.Position), attachment.CFrame
	local offsets = {vector.zero, vector.new(0.2, 0, 0), vector.new(-0.2, 0, 0), vector.new(0, 0.2, 0), vector.new(0, -0.2, 0)}
	local function clear(pose)
		local endpoint = (pose * muzzle).Position
		for _, offset in offsets do
			local origin = pose * (headLocal + offset)
			local direction = endpoint - origin
			if direction.Magnitude > 0.001 and workspace:Raycast(origin, direction * ((direction.Magnitude + 0.6) / direction.Magnitude), ray) then return false end
		end
		return true
	end
	for index = 0, 7 do
		local pose = frame.new(position) * frame.Angles(0, index * math.pi / 4, 0)
		if clear(pose) and clear(pose * frame.Angles(0, math.rad(25), 0)) and clear(pose * frame.Angles(0, -math.rad(25), 0)) then return true end
	end
	return false
end

function FarmMotion.findShootSpot(context, root, position, startDepth)
	local vector = context.environment.Vector3
	local minimum = context.workspace.FallenPartsDestroyHeight + 25
	local depth = startDepth or 0
	for offset = depth, depth + 48, 2 do
		local candidate = position - vector.new(0, offset, 0)
		if candidate.Y < minimum then break end
		if FarmMotion.canShootAt(context, candidate, root) then return candidate, offset end
	end
	return position - vector.new(0, depth, 0), depth
end

function FarmMotion.shootSpot(context, state, motion, root, now)
	local spot = motion.spot
	local vector = context.environment.Vector3
	if not spot or vector.new(spot.X - root.Position.X, 0, spot.Z - root.Position.Z).Magnitude > 10 then
		motion.spotBase = FarmMotion.hideAt(context, state, motion, root, now)
		motion.spot, motion.spotD = FarmMotion.findShootSpot(context, root, motion.spotBase, 0)
		motion.spotT = now
		return motion.spot
	end
	if (spot - root.Position).Magnitude < 1 and now - (motion.spotT or 0) > 1 then
		local guard = rawget(context.environment.getgenv(), "__pulseMuzzle")
		if type(guard) == "table" and guard.at and now - guard.at < 0.2 and guard.blocked then
			motion.spot, motion.spotD = FarmMotion.findShootSpot(context, root, motion.spotBase or spot, (motion.spotD or 0) + 2)
		end
		motion.spotT = now
	end
	return motion.spot
end

function FarmMotion.skip(context, state, coin)
	state.skipped[coin] = true
	context.environment.task.delay(30, function() state.skipped[coin] = nil end)
end

function FarmMotion.underMap(context, state, motion, options)
	local now, root, velocity, gyro, character = options.time, options.root, options.velocity, options.gyro, options.character
	local murderer, gun = options.murderer, options.gun
	local speed = math.clamp(tonumber(context.configuration.farm.speed) or 21, 8, 22)
	local coin = motion.coin
	if gun then coin = gun end
	if not gun then
		if coin and not FarmState.validCoin(context, state, coin) then
			if motion.phase == "rise" and motion.riseFrom then
				table.insert(motion.learn, coin.Position.Y - root.Position.Y)
				if #motion.learn > 60 then table.remove(motion.learn, 1) end
			end
			motion.got, motion.coin, motion.phase, coin = motion.got + 1, nil, "go", nil
		end
		if not coin or motion.phase == "go" and now - (motion.pickT or 0) > 0.3 then
			motion.pickT = now
			local candidate, score = FarmMotion.choose(context, state, motion, root.Position, murderer, now)
			if candidate and candidate ~= coin then
				local current
				if coin and FarmState.validCoin(context, state, coin) then
					local info = FarmMotion.coinInfo(context, state, motion, coin, now)
					current = (info.hide - root.Position).Magnitude + math.max(0, coin.Position.Y - info.hide.Y) * 1.2
						+ (info.exposed and 25 or 0) + FarmMotion.murderPenalty(state, coin.Position, murderer)
				end
				if not current or score + 6 < current then
					motion.coin, motion.phase, motion.goT, motion.tries, coin = candidate, "go", now, motion.tries + 1, candidate
				end
			end
		end
	end
	if not coin then
		FarmMotion.move(context, root, velocity, gyro, FarmMotion.hideAt(context, state, motion, root, now), options.delta, speed)
		motion.phase = "wait"
		return
	end
	local info = FarmMotion.coinInfo(context, state, motion, coin, now)
	if motion.phase ~= "rise" then
		if motion.phase ~= "go" then motion.phase, motion.goT = "go", now end
		local force = state.firstJump and coin ~= gun
		if force then state.firstJump = false end
		FarmMotion.jump(context, motion, root, info.hide, force, now)
		local distance = FarmMotion.move(context, root, velocity, gyro, info.hide, options.delta, speed)
		if distance < 0.7 then motion.phase, motion.riseFrom, motion.dwellFrom = "rise", now, nil; return end
		if now - (motion.goT or now) > distance / math.max(speed, 1) + 4 then
			FarmMotion.skip(context, state, coin)
			motion.coin, motion.gunTarget, motion.fails, motion.phase = nil, nil, motion.fails + 1, "go"
		end
		return
	end
	if murderer and coin ~= gun then
		local distance = context.environment.Vector3.new(coin.Position.X - murderer.X, 0, coin.Position.Z - murderer.Z).Magnitude
		if distance < 12 then motion.skip[coin], motion.coin, motion.phase = now + 3, nil, "go"; return end
	end
	local attempts = motion.att[coin] or 0
	local gap = math.max(0.6, motion.gapT - 0.45 * attempts)
	local destination = context.environment.Vector3.new(info.hide.X, math.max(coin.Position.Y - gap, info.hide.Y), info.hide.Z)
	local distance = FarmMotion.move(context, root, velocity, gyro, destination, options.delta, speed)
	if now - (motion.touchAt or 0) >= 1 / 30 then motion.touchAt = now; FarmMotion.touch(context, character, coin) end
	if distance < 0.3 then
		motion.dwellFrom = motion.dwellFrom or now
		if now - motion.dwellFrom >= 0.1 + (coin == gun and 0.25 or 0) then
			motion.att[coin] = attempts + 1
			if attempts + 1 >= 4 then FarmMotion.skip(context, state, coin); motion.fails += 1 end
			if attempts + 1 < 4 then motion.pending[coin] = {t = now + 0.9, gap = gap} end
			if coin == gun then motion.gunTarget = nil end
			motion.coin, motion.dwellFrom, motion.phase = nil, nil, "go"
			return
		end
	end
	if now - (motion.riseFrom or now) > 1.4 then
		FarmMotion.skip(context, state, coin)
		if coin == gun then motion.gunTarget = nil end
		motion.coin, motion.fails, motion.phase = nil, motion.fails + 1, "go"
	end
end

function FarmMotion.fast(context, state, motion, options)
	local now, root, velocity, gyro, character = options.time, options.root, options.velocity, options.gyro, options.character
	local gun, murderer = options.gun, options.murderer
	local speed = math.clamp(tonumber(context.configuration.farm.fastSpeed) or 24, 8, 24)
	local coin = motion.coin
	if coin and coin ~= gun and (not FarmState.validCoin(context, state, coin) or motion.pending[coin]) then
		if not FarmState.validCoin(context, state, coin) then motion.got += 1 end
		motion.coin, coin = nil, nil
	end
	local target = gun
	if not target then
		if not coin or now - (motion.pickT or 0) > 0.3 then
			motion.pickT = now
			local best, lowest
			for _, container in FarmMotion.containers(context, state, motion, now) do
				if not container.Parent then continue end
				for _, candidate in container:GetChildren() do
					if not FarmState.validCoin(context, state, candidate) or (motion.skip[candidate] or 0) >= now or motion.pending[candidate] then continue end
					local score = (candidate.Position - root.Position).Magnitude + FarmMotion.murderPenalty(state, candidate.Position, murderer)
					if not lowest or score < lowest then best, lowest = candidate, score end
				end
			end
			if best and state.avoidMurder and lowest >= 1000000 then best = nil end
			if best and best ~= coin and (not coin or lowest + 6 < (coin.Position - root.Position).Magnitude + FarmMotion.murderPenalty(state, coin.Position, murderer)) then
				motion.coin, motion.goT, motion.tries, coin = best, now, motion.tries + 1, best
			end
		end
		target = coin
	end
	if not target then velocity.Velocity, motion.phase = context.environment.Vector3.zero, "wait"; return end
	motion.phase = "go"
	local force = state.firstJump and target ~= gun
	if force then state.firstJump = false end
	FarmMotion.jump(context, motion, root, target.Position, force, now)
	local distance = FarmMotion.move(context, root, velocity, gyro, target.Position, options.delta, speed)
	if distance < 4 then FarmMotion.touch(context, character, target) end
	if distance < 1 then
		motion.dwellFrom = motion.dwellFrom or now
		if now - motion.dwellFrom >= (target == gun and 0.25 or 0.05) then
			if target == gun then motion.gunTarget = nil end
			if target ~= gun then motion.pending[target] = {t = now + 0.9} end
			motion.coin, motion.dwellFrom = nil, nil
		end
		return
	end
	motion.dwellFrom = nil
	if now - (motion.goT or now) > distance / speed + 4 then
		FarmMotion.skip(context, state, target)
		if target == gun then motion.gunTarget = nil end
		motion.coin, motion.fails = nil, motion.fails + 1
	end
end

function FarmMotion.setTouch(context, state, motion, target, now)
	local teleport = motion.tp
	teleport.ct = teleport.ct or {}
	for coin in teleport.ct do
		if coin ~= target and coin.Parent then continue end
		if coin.Parent then pcall(function() coin.CanTouch = true end) end
		teleport.ct[coin] = nil
	end
	for _, container in FarmMotion.containers(context, state, motion, now) do
		if not container.Parent then continue end
		for _, coin in container:GetChildren() do
			if coin == target or teleport.ct[coin] or not coin:IsA("BasePart") or not coin.CanTouch or not coin:FindFirstChild("TouchInterest") then continue end
			teleport.ct[coin] = true
			pcall(function() coin.CanTouch = false end)
		end
	end
end

function FarmMotion.restoreTouch(motion)
	local teleport = motion.tp
	if not teleport or not teleport.ct then return end
	for coin in teleport.ct do
		if coin.Parent and not coin:GetAttribute("Collected") then pcall(function() coin.CanTouch = true end) end
	end
	teleport.ct = nil
end

function FarmMotion.teleportPick(context, state, motion, now, murderer, previous, current)
	local coins, teleport = {}, motion.tp
	for _, container in FarmMotion.containers(context, state, motion, now) do
		if not container.Parent then continue end
		for _, coin in container:GetChildren() do
			if not FarmState.validCoin(context, state, coin) or (motion.skip[coin] or 0) >= now then continue end
			table.insert(coins, coin)
			teleport.valid = teleport.valid or setmetatable({}, {__mode = "k"})
			teleport.valid[coin] = true
		end
	end
	if #coins == 0 then return nil end
	if previous and (coins[1].Position - previous).Magnitude > 1000 then previous = nil end
	local best, lowest, currentScore
	for _, coin in coins do
		local nearest, second, third = math.huge, math.huge, math.huge
		for _, other in coins do
			if other == coin then continue end
			local distance = (other.Position - coin.Position).Magnitude
			if distance < nearest then nearest, second, third = distance, nearest, second; continue end
			if distance < second then second, third = distance, second; continue end
			if distance < third then third = distance end
		end
		local score = previous and ((coin.Position - previous).Magnitude + 0.25 * (math.min(nearest, 120) + math.min(second, 120)))
			or math.min(nearest, 300) + math.min(second, 300) + math.min(third, 300)
		score += FarmMotion.murderPenalty(state, coin.Position, murderer)
		if coin == current then currentScore = score end
		if not lowest or score < lowest then best, lowest = coin, score end
	end
	if best and state.avoidMurder and lowest >= 1000000 then return nil end
	return best, lowest, currentScore
end

function FarmMotion.home(context, motion, now)
	if motion.homeCF and now - (motion.homeAt or 0) < 5 then return motion.homeCF end
	motion.homeAt, motion.homeCF = now, nil
	local lobby = context.workspace:FindFirstChild("Lobby")
	local spawns = lobby and lobby:FindFirstChild("Spawns")
	if not spawns then return nil end
	local total, count = context.environment.Vector3.zero, 0
	for _, part in spawns:GetDescendants() do
		if part:IsA("BasePart") then total, count = total + part.Position, count + 1 end
	end
	if count > 0 then motion.homeCF = context.environment.CFrame.new(total / count + context.environment.Vector3.new(0, 4, 0)) end
	return motion.homeCF
end

function FarmMotion.teleportReference(context, state, motion, root, now, tickTime)
	local teleport = motion.tp
	if teleport.lastAt == nil then
		teleport.lastAt = motion.evAt or 0
		if motion.evAt and now - motion.evAt < 20 then teleport.lastPos = motion.evPos end
		teleport.seenEv = motion.evAt or 0
		if not motion.evAt and state.coinsStarted > 0 and tickTime - state.coinsStarted > 5 then teleport.cold = now + 2.5 end
	end
	local event = motion.evAt
	if not event or event <= (teleport.seenEv or 0) then return end
	teleport.seenEv = event
	if event <= teleport.lastAt then return end
	teleport.lastAt = event
	if teleport.touchC and event - (teleport.touchT or 0) < 1.5 then teleport.lastPos = teleport.touchC.Position; return end
	local position, nearest, selected = root and root.Position, nil, nil
	for coin in teleport.valid or {} do
		if not coin.Parent or not coin:GetAttribute("Collected") or not position then continue end
		local distance = (coin.Position - position).Magnitude
		if distance < 14 and (not nearest or distance < nearest) then nearest, selected = distance, coin end
	end
	if selected then teleport.lastPos = selected.Position; return end
	teleport.lastPos, teleport.cold = nil, event + 3
end

function FarmMotion.teleport(context, state, motion, options)
	local now, root, velocity, character, gun = options.time, options.root, options.velocity, options.character, options.gun
	local teleport = motion.tp
	local home = FarmMotion.home(context, motion, now) or teleport.start
	teleport.start = teleport.start or root.CFrame
	velocity.Velocity = context.environment.Vector3.zero
	FarmMotion.teleportReference(context, state, motion, root, now, options.tickTime)
	local function returnHome(phase, force)
		motion.phase = phase
		if home and (force or (root.Position - home.Position).Magnitude > 6) then root.CFrame = home end
		root.AssemblyLinearVelocity = context.environment.Vector3.zero
	end
	if teleport.cold and not gun and now < teleport.cold then returnHome("wait"); return end
	local coin = teleport.coin
	if coin then
		local isGun = coin == gun
		local collected = isGun and (coin.Parent == nil or options.hasGun) or not isGun and not FarmState.validCoin(context, state, coin)
		local done = false
		if collected then
			if isGun then motion.gunTarget = nil end
			if not isGun and coin:GetAttribute("Collected") then motion.got += 1 end
			if not isGun and teleport.touching then
				if now > teleport.lastAt then teleport.lastAt = now end
				teleport.lastPos = coin.Position
				local latency = now - (teleport.at or now)
				if latency > 0 and latency < 0.6 then motion.tpLat = math.clamp((motion.tpLat or 0.1) * 0.8 + latency * 0.2, 0.03, 0.3) end
			end
			done = true
		end
		if not collected then
			if teleport.touching and now - teleport.at > 0.9 then
				FarmMotion.skip(context, state, coin)
				motion.fails += 1
				if isGun then motion.gunTarget = nil end
				done = true
			end
			if gun and not isGun then done = true end
		end
		if done then teleport.coin, teleport.touching = nil, false; returnHome("home", true); return end
	end
	if options.full and not gun then returnHome("full"); return end
	if not teleport.coin then
		local target = gun or FarmMotion.teleportPick(context, state, motion, now, options.murderer, teleport.lastPos)
		if not target then returnHome("home"); return end
		teleport.coin, teleport.touching, teleport.at = target, false, now
		teleport.hide = target == gun and target.Position or FarmMotion.coinInfo(context, state, motion, target, now).hide
		motion.tries += 1
		FarmMotion.setTouch(context, state, motion, target, now)
		teleport.ctAt = now
	end
	coin = teleport.coin
	if now - (teleport.ctAt or 0) > 0.1 then
		teleport.ctAt = now
		FarmMotion.setTouch(context, state, motion, coin, now)
	end
	local isGun = coin == gun
	if not isGun and not teleport.touching and now - (teleport.rpAt or 0) > 0.2 then
		teleport.rpAt = now
		local target, score, currentScore = FarmMotion.teleportPick(context, state, motion, now, options.murderer, teleport.lastPos, coin)
		if target and target ~= coin and (not currentScore or score + 2 < currentScore) then
			coin, teleport.coin, teleport.at = target, target, now
			teleport.hide = FarmMotion.coinInfo(context, state, motion, target, now).hide
			FarmMotion.setTouch(context, state, motion, target, now)
			teleport.ctAt = now
		end
	end
	local arrival = now
	if not isGun and teleport.lastPos and (coin.Position - teleport.lastPos).Magnitude < 1000 then
		local speed = math.clamp(tonumber(context.configuration.farm.tpSpeed) or 25, 10, 26)
		arrival = math.max(teleport.lastAt, teleport.lastAt + (coin.Position - teleport.lastPos).Magnitude / speed - 0.4 * (motion.tpLat or 0.1))
	end
	if not teleport.touching and now < arrival - 0.35 then returnHome("wait"); return end
	root.CFrame = context.environment.CFrame.new(teleport.hide)
	root.AssemblyLinearVelocity = context.environment.Vector3.zero
	if teleport.touching or now >= arrival then
		if not teleport.touching then teleport.touching, teleport.at = true, now end
		teleport.touchC, teleport.touchT = coin, now
		FarmMotion.touch(context, character, coin)
		motion.phase = "tp"
		return
	end
	motion.phase = "wait"
end

function FarmMotion.ensureMover(context, root, humanoid, withoutGyro)
	local environment = context.environment
	local velocity = root:FindFirstChild("PulseFarmBV")
	if not velocity then
		velocity = environment.Instance.new("BodyVelocity")
		velocity.Name, velocity.MaxForce, velocity.P = "PulseFarmBV", environment.Vector3.new(9e9, 9e9, 9e9), 30000
		velocity.Velocity = environment.Vector3.zero
		velocity:SetAttribute("PulseMine", true)
		velocity.Parent = root
	end
	local gyro = root:FindFirstChild("PulseFarmBG")
	if withoutGyro and gyro then pcall(function() gyro:Destroy() end); gyro = nil end
	if not withoutGyro and not gyro then
		gyro = environment.Instance.new("BodyGyro")
		gyro.Name, gyro.MaxTorque, gyro.P = "PulseFarmBG", environment.Vector3.new(9e9, 9e9, 9e9), 30000
		gyro:SetAttribute("PulseMine", true)
		gyro.CFrame = environment.CFrame.lookAt(root.Position, root.Position + environment.Vector3.new(root.CFrame.LookVector.X, 0, root.CFrame.LookVector.Z + 1e-4))
		gyro.Parent = root
	end
	if humanoid and not humanoid.PlatformStand then pcall(function() humanoid.PlatformStand = true end) end
	return velocity, gyro
end

function FarmMotion.move(context, root, velocity, gyro, destination, delta, speed)
	local direction = destination - root.Position
	local distance = direction.Magnitude
	velocity.Velocity = distance < 0.05 and context.environment.Vector3.zero or direction.Unit * math.min(speed, distance / math.max(delta, 1 / 60))
	local horizontal = context.environment.Vector3.new(direction.X, 0, direction.Z)
	if gyro and horizontal.Magnitude > 0.4 then gyro.CFrame = context.environment.CFrame.lookAt(root.Position, root.Position + horizontal.Unit) end
	return distance
end

function FarmMotion.reset(context, motion, character)
	pcall(function()
		local root = character and character:FindFirstChild("HumanoidRootPart")
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		if root then
			for _, name in {"PulseFarmBV", "PulseFarmBG"} do local object = root:FindFirstChild(name); if object then object:Destroy() end end
		end
		if humanoid and humanoid.PlatformStand then
			humanoid.PlatformStand = false
			if humanoid.Health > 0 then humanoid:ChangeState(context.environment.Enum.HumanoidStateType.GettingUp) end
		end
	end)
	if motion.ctRestore then pcall(motion.ctRestore) end
	motion.phase, motion.coin, motion.hold = "idle", nil, nil
end

function FarmMotion.updatePending(context, state, motion, now)
	for coin, entry in motion.pending do
		if type(entry) ~= "table" then continue end
		if FarmState.validCoin(context, state, coin) then
			if now > entry.t then
				motion.pending[coin] = nil
				motion.gapT, motion.miss = math.max(1, motion.gapT - 0.25), (motion.miss or 0) + 1
			end
			continue
		end
		motion.pending[coin] = nil
		motion.gapT, motion.ok = math.min(6, motion.gapT + 0.08), (motion.ok or 0) + 1
	end
end

local collisionSnapshot = setmetatable({}, {__mode = "k"})
local noclipCharacter = nil

local function setFarmCollision(enabled)
    local character = LocalPlayer.Character
    if noclipCharacter and noclipCharacter ~= character then
        for part, original in pairs(collisionSnapshot) do
            if part and part.Parent then pcall(function() part.CanCollide = original end) end
            collisionSnapshot[part] = nil
        end
        noclipCharacter = nil
    end
    if not character then return end
    if enabled then
        noclipCharacter = character
        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BasePart") then
                if collisionSnapshot[part] == nil then collisionSnapshot[part] = part.CanCollide end
                if part.CanCollide then part.CanCollide = false end
            end
        end
    else
        for part, original in pairs(collisionSnapshot) do
            if part and part.Parent then pcall(function() part.CanCollide = original end) end
            collisionSnapshot[part] = nil
        end
        noclipCharacter = nil
    end
end

local roleSource = nil
local roleCache = {}
local roleCacheAt = 0

local function getRoleData()
    local now = os.clock()
    if now - roleCacheAt < 0.25 then return roleCache end
    roleCacheAt = now
    if not roleSource then
        local function tryModule(folder)
            local module = folder and folder:FindFirstChild("CurrentRoundClient")
            if not module or not module:IsA("ModuleScript") then return nil end
            local ok, loaded = pcall(require, module)
            if ok and type(loaded) == "table" and type(loaded.GetLatestPlayerData) == "function" then
                return loaded
            end
            return nil
        end
        pcall(function()
            local modules = context.storage:FindFirstChild("Modules")
            roleSource = tryModule(modules)
                or tryModule(context.storage)
                or tryModule(modules and modules:FindFirstChild("Client"))
        end)
    end
    if roleSource then
        local ok, roles = pcall(function() return roleSource.GetLatestPlayerData() end)
        if ok and type(roles) == "table" then roleCache = roles end
    end
    return roleCache
end

local function getMurdererPosition()
    local roles = getRoleData()
    for _, player in ipairs(Players:GetPlayers()) do
        local role = roles[player.Name]
        if player ~= LocalPlayer and role and role.Role == "Murderer"
            and role.Dead ~= true and role.Killed ~= true then
            local character = player.Character
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if humanoid and humanoid.Health > 0 and root then return root.Position end
        end
    end
    return nil
end

local function farmStep(delta)
    if destroyed then return end
    local now = os.clock()
    local character, humanoid, root = getCharacter()
    local containers = FarmState.containers(context, state, now)
    local hasCoins = false
    for _, container in containers do
        if container.Parent and #container:GetChildren() > 0 then hasCoins = true; break end
    end

    if hasCoins then
        noCoinsSince = nil
        if not state.round then
            state.round = true
            state.firstJump = true
            state.coinsStarted = tick()
        end
    elseif state.round then
        noCoinsSince = noCoinsSince or now
        if now - noCoinsSince > 8 then
            FarmState.roundEnd(state)
            state.active = false
            resetMotion()
        end
    end

    if not running or not state.round or not character or not humanoid or not root then
        state.active = false
        if motion.phase ~= "idle" then FarmMotion.reset(context, motion, LocalPlayer.Character) end
        return
    end

    state.active = true
    FarmMotion.updatePending(context, state, motion, now)
    local full = FarmState.isFull(context, state, tick())
    local method = context.configuration.farm.method
    local velocity, gyro = FarmMotion.ensureMover(context, root, humanoid, method == "Teleport")

    if full then
        motion.phase = "full"
        motion.coin = nil
        if state.respawnOnFull and not state.forcedFull and not state.respawnPending
            and tick() - (state.lastRespawn or 0) > 3 then
            state.respawnPending = true
            task.delay(0.35, function()
                state.respawnPending = false
                if not running or not FarmState.isFull(context, state, tick()) or state.forcedFull then return end
                local current = LocalPlayer.Character
                local currentHumanoid = current and current:FindFirstChildOfClass("Humanoid")
                if currentHumanoid and currentHumanoid.Health > 0 then
                    state.active = false
                    state.lastRespawn = tick()
                    state.forcedFull = true
                    table.clear(state.full)
                    currentHumanoid.Health = 0
                end
            end)
            return
        end
        local hold = FarmMotion.hideAt(context, state, motion, root, now)
        FarmMotion.move(context, root, velocity, gyro, hold, delta, math.clamp(speed, 8, 22))
        return
    end

    local options = {
        time = now,
        tickTime = tick(),
        root = root,
        velocity = velocity,
        gyro = gyro,
        character = character,
        delta = delta,
        gun = nil,
        murderer = state.avoidMurder and getMurdererPosition() or nil,
        full = false,
        hasGun = false,
    }

    if method == "Teleport" then
        state.firstJump = false
        FarmMotion.teleport(context, state, motion, options)
    elseif method == "Fast" then
        FarmMotion.fast(context, state, motion, options)
    else
        FarmMotion.underMap(context, state, motion, options)
    end

    if now - lastStats > 0.5 then
        lastStats = now
        updateUi()
    end
end
