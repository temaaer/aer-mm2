function AER:Create(className, properties, children)
    local instance = Instance.new(className)
    for prop, value in pairs(properties) do
        instance[prop] = value
    end
    if children then
        for _, child in pairs(children) do
            child.Parent = instance
        end
    end
    return instance
end

function AER:Round(instance, radius)
    self:Create("UICorner", {CornerRadius = UDim.new(0, radius)}, {Parent = instance})
end

function AER:GetTween(instance, duration, properties, style, direction)
    return TweenService:Create(instance, TweenInfo.new(duration or 0.3, style or Enum.EasingStyle.Quint, direction or Enum.EasingDirection.Out), properties)
end

--// Anti-Fling (from file)
local function AntiFling()
    local connection
    connection = LocalPlayer.CharacterAdded:Connect(function(char)
        local humanoid = char:WaitForChild("Humanoid")
        local root = char:WaitForChild("HumanoidRootPart")
        local oldVelocity
        local antiFlingConnection = RunService.Heartbeat:Connect(function()
            if not Config.AutoFarm then antiFlingConnection:Disconnect() return end
            if root and root.Parent then
                if root.AssemblyAngularVelocity.Magnitude > 50 then
                    root.AssemblyAngularVelocity = Vector3.zero
                end
            end
        end)
        table.insert(AER.Connections, antiFlingConnection)
    end)
    table.insert(AER.Connections, connection)
end

--// Bhop (from file)
local bhopState = {speed = 0, landedAt = 0, wasGrounded = true, reserve = 0, chainUntil = 0}
local function Bhop(delta)
    if not Config.AutoFarm or not Config.Bhop then
        if bhopState.reserve > 0 then
            bhopState.speed, bhopState.reserve, bhopState.chainUntil = 0, 0, 0
        end
        return
    end
    local character = LocalPlayer.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local stateTypes = Enum.HumanoidStateType
    if humanoid and root and humanoid.Health > 0 then
        local status = humanoid:GetState()
        if status == stateTypes.Climbing or status == stateTypes.Freefall then return end
        local grounded = humanoid.FloorMaterial ~= Enum.Material.Air and status ~= stateTypes.Jumping
        local now, velocity = os.clock(), root.AssemblyLinearVelocity
        local power, walkSpeed = Config.BhopPower, humanoid.WalkSpeed
        if grounded and not bhopState.wasGrounded then bhopState.landedAt = now end
        if not grounded and bhopState.wasGrounded and velocity.Y > 2 and humanoid.MoveDirection.Magnitude > 0.1 then
            bhopState.speed = bhopState.speed > 0 and now - bhopState.landedAt <= 0.25 and bhopState.speed + 2 + power * 1.2 or math.max(bhopState.speed, walkSpeed + 2 + power)
        end
        if grounded and now - bhopState.landedAt > 0.3 then bhopState.speed = 0 end
        local movement = humanoid.MoveDirection
        if bhopState.speed > 0 and movement.Magnitude > 0.1 then
            bhopState.speed = math.min(bhopState.speed, math.max(walkSpeed * (2 + 0.4 * power), 100))
            if not grounded or now - bhopState.landedAt <= 0.25 then
                local direction = Vector3.new(movement.X, 0, movement.Z).Unit
                root.AssemblyLinearVelocity = Vector3.new(direction.X * bhopState.speed, velocity.Y, direction.Z * bhopState.speed)
            end
        end
        bhopState.wasGrounded = grounded
        if bhopState.speed > 0 then bhopState.reserve = math.max(bhopState.reserve, bhopState.speed + 2 + power * 1.2); bhopState.chainUntil = now + 0.5 end
    else
        bhopState.wasGrounded, bhopState.speed, bhopState.reserve, bhopState.chainUntil = true, 0, 0, 0
    end
end
RunService.Heartbeat:Connect(Bhop)

--// Auto Shoot Murderer (from file)
local function getPlayersWithRole(roleName)
    local roles = {}
    local storage = game:GetService("ReplicatedStorage")
    local modules = storage:FindFirstChild("Modules")
    local source = modules and modules:FindFirstChild("Client") and modules.Client:FindFirstChild("CurrentRoundClient")
    if source and source:IsA("ModuleScript") then
        local ok, module = pcall(require, source)
        if ok and type(module) == "table" and type(module.GetLatestPlayerData) == "function" then
            local success, data = pcall(module.GetLatestPlayerData)
            if success and type(data) == "table" then
                for playerName, roleData in pairs(data) do
                    if roleData.Role == roleName and not roleData.Dead and not roleData.Killed then
                        table.insert(roles, Players:FindFirstChild(playerName))
                    end
                end
            end
        end
    end
    if #roles == 0 then -- Fallback method
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then
                for _, tool in ipairs(p.Character:GetChildren()) do
                    if tool:IsA("Tool") and string.find(string.lower(tool.Name), string.lower(roleName)) then
                        table.insert(roles, p)
                        break
                    end
                end
            end
        end
    end
    return roles
end

local autoShootConnection
local function setupAutoShoot()
    if autoShootConnection then autoShootConnection:Disconnect() end
    if not Config.AutoFarm or not Config.AutoShoot then return end
    
    autoShootConnection = RunService.RenderStepped:Connect(function()
        local murderers = getPlayersWithRole("Murderer")
        if #murderers == 0 then return end
        
        local localChar = LocalPlayer.Character
        if not localChar then return end
        local gun = localChar:FindFirstChild("Gun")
        if not gun then return end
        
        local murderer = murderers[1]
        local murdererChar = murderer and murderer.Character
        local murdererRoot = murdererChar and murdererChar:FindFirstChild("HumanoidRootPart")
        
        if murdererRoot then
            -- Aim
            local camera = Workspace.CurrentCamera
            camera.CFrame = CFrame.new(camera.CFrame.Position, murdererRoot.Position)
            
            -- Shoot
            local gunRemote = gun:FindFirstChild("Fire")
            if gunRemote then
                gunRemote:FireServer(murdererRoot.Position, murdererChar.Head)
            end
        end
    end)
end

--// Auto Fling Sheriff (from file)
local flingConnection
local function setupAutoFling()
    if flingConnection then flingConnection:Disconnect() end
    if not Config.AutoFarm or not Config.AutoFlingSheriff then return end
    
    flingConnection = RunService.Heartbeat:Connect(function()
        local sheriffs = getPlayersWithRole("Sheriff")
        if #sheriffs == 0 then return end
        
        local target = sheriffs[1]
        local targetChar = target and target.Character
        local targetRoot = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
        local localChar = LocalPlayer.Character
        local localRoot = localChar and localChar:FindFirstChild("HumanoidRootPart")

        if targetRoot and localRoot and (targetRoot.Position - localRoot.Position).Magnitude < 10 then
            -- Simplified fling logic
            for _, part in ipairs(targetChar:GetDescendants()) do
                if part:IsA("BasePart") then
                    local ok = pcall(function() part.Velocity = Vector3.new(math.huge, math.huge, math.huge) end)
                    if ok then break end
                end
            end
        end
    end)
end

--// Kill Aura (Aggressive method)
local killAuraConnection
local function setupKillAura()
    if killAuraConnection then killAuraConnection:Disconnect() end
    if not Config.AutoFarm or Config.FarmMethod ~= "Aggressive" or not Config.KillAura then return end

    killAuraConnection = RunService.Heartbeat:Connect(function()
        local localChar = LocalPlayer.Character
        if not localChar then return end
        
        -- Prioritize Murderer, then others
        local targets = getPlayersWithRole("Murderer")
        if #targets == 0 then
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                    table.insert(targets, player)
                end
            end
        end
        
        for _, target in ipairs(targets) do
            local targetChar = target.Character
            if not targetChar then continue end
            local targetRoot = targetChar:FindFirstChild("HumanoidRootPart")
            local targetHum = targetChar:FindFirstChildOfClass("Humanoid")
            if not targetRoot or not targetHum or targetHum.Health <= 0 then continue end
            
            if (localChar.HumanoidRootPart.Position - targetRoot.Position).Magnitude < 15 then
                -- Simple touch-based damage
                firetouchinterest(localChar.HumanoidRootPart, targetRoot, 0)
                task.wait()
                firetouchinterest(localChar.HumanoidRootPart, targetRoot, 1)
            end
        end
    end)
end


--// Main Logic Handler
local function updateAutoFarm()
    for _, conn in ipairs(AER.Connections) do
        if conn then conn:Disconnect() end
    end
    AER.Connections = {}
    
    if Config.AutoFarm then
        AntiFling()
        setupAutoShoot()
        setupAutoFling()
        setupKillAura()
    end
end
